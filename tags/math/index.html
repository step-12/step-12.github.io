        JS Math Reference

        October 3, 2026
        Javascript, Math
        // ============================================================================
// JavaScript Math Reference
// ============================================================================
//
// A compact reference for common mathematical operations in JavaScript.
//
// ============================================================================



// ============================================================================
// ABSOLUTE VALUE
// ============================================================================

Math.abs(5);             // 5
Math.abs(-5);            // 5
Math.abs(0);             // 0

Math.abs(10 - 15);       // 5



// ============================================================================
// MINIMUM AND MAXIMUM
// ============================================================================

Math.min(2, 5);          // 2
Math.min(10, 3, 7);      // 3

Math.max(2, 5);          // 5
Math.max(10, 3, 7);      // 10

Math.min(-5, -2);        // -5
Math.max(-5, -2);        // -2



// ============================================================================
// SQUARE ROOT
// ============================================================================

Math.sqrt(9);            // 3
Math.sqrt(16);           // 4
Math.sqrt(2);            // approximately 1.414

// The Pythagorean theorem:
//
//     c = √(a² + b²)

Math.sqrt(3 ** 2 + 4 ** 2);   // 5



// ============================================================================
// POWERS
// ============================================================================

Math.pow(2, 3);          // 8
Math.pow(5, 2);          // 25

// The ** operator does the same thing.

2 ** 3;                  // 8
5 ** 2;                  // 25



// ============================================================================
// ROUNDING
// ============================================================================

Math.floor(3.9);         // 3
Math.floor(3.1);         // 3
Math.floor(-3.1);        // -4

Math.ceil(3.1);          // 4
Math.ceil(3.9);          // 4
Math.ceil(-3.1);         // -3

Math.round(3.1);         // 3
Math.round(3.5);         // 4
Math.round(3.9);         // 4

Math.trunc(3.9);         // 3
Math.trunc(-3.9);        // -3



// ============================================================================
// RANDOM NUMBERS
// ============================================================================

Math.random();           // A random number from 0 up to, but not including, 1.

Math.random();           // 0.374...
Math.random();           // 0.821...
Math.random();           // 0.092...



// ----------------------------------------------------------------------------
// Random integer
// ----------------------------------------------------------------------------

// A random integer from 0 through 9:

Math.floor(Math.random() * 10);

// A random integer from 0 through 99:

Math.floor(Math.random() * 100);

// A random integer from 1 through 10:

Math.floor(Math.random() * 10) + 1;



// ============================================================================
// CONSTANTS
// ============================================================================

Math.PI;                 // approximately 3.14159
Math.E;                  // approximately 2.71828

Math.SQRT2;              // approximately 1.41421
Math.SQRT1_2;            // approximately 0.70711



// ============================================================================
// TRIGONOMETRY
// ============================================================================
//
// JavaScript uses radians, not degrees.
//
//     180° = π radians
//      90° = π / 2 radians
//
// ============================================================================

Math.sin(0);             // 0
Math.cos(0);             // 1
Math.tan(0);             // 0

Math.sin(Math.PI / 2);   // 1
Math.cos(Math.PI);       // -1



// ----------------------------------------------------------------------------
// Converting degrees to radians
// ----------------------------------------------------------------------------

const degrees = 90;
const radians = degrees * Math.PI / 180;

Math.sin(radians);       // 1



// ----------------------------------------------------------------------------
// Converting radians to degrees
// ----------------------------------------------------------------------------

const radians = Math.PI;
const degrees = radians * 180 / Math.PI;

degrees;                 // 180



// ============================================================================
// INVERSE TRIGONOMETRY
// ============================================================================

Math.asin(1);            // approximately π / 2
Math.acos(1);            // 0
Math.atan(1);            // approximately π / 4

Math.atan2(1, 1);        // approximately π / 4



// ============================================================================
// LOGARITHMS AND EXPONENTS
// ============================================================================

Math.exp(1);             // e

Math.log(Math.E);        // 1
Math.log(1);             // 0

Math.log10(100);         // 2
Math.log2(8);            // 3



// ============================================================================
// SIGN
// ============================================================================

Math.sign(10);           // 1
Math.sign(-10);          // -1
Math.sign(0);            // 0



// ============================================================================
// USEFUL COMBINATIONS
// ============================================================================

// Distance between two points on a number line:

Math.abs(a - b);


// Distance between two points (x1, y1) and (x2, y2):

Math.sqrt(
    (x2 - x1) ** 2 +
    (y2 - y1) ** 2
);


// Clamp a value between a minimum and maximum:

Math.min(Math.max(value, minimum), maximum);


// Round a number to the nearest integer:

Math.round(value);


// Round down:

Math.floor(value);


// Round up:

Math.ceil(value);

        Functions — A Little Bit of Function#

        1. What Is a Function?#

        A function is a rule that takes an input and produces an output.

        For example:

        f(x) = 2x + 3

        Give it 5:

        f(5) = 2(5) + 3
     = 13

        So we can think of a function as:

        input → function → output

        2. Evaluating Functions#

        If:

        f(x) = x² + 1

        then:

        f(2) = 2² + 1
     = 5

        The x is simply replaced by the input.

        Logic: The Machinery of Reasoning#

        You already know more logic than you think.

        Every time you say:

        “If it is raining, I should carry an umbrella.”

        or:

        “I will have tea if there is milk and sugar.”

        or:

        “I will go out if it is Saturday or Sunday.”

        you are using logic.

        You don’t need a computer, a formula, or a mathematics degree to use it. Logic is simply a way of describing how statements combine and how we reason from one statement to another.
