    // ============================================================================
// JavaScript Language Reference
// ============================================================================
//
// A compact reference of JavaScript syntax and expressions.
// Every example in this file is valid JavaScript.
//
// ============================================================================



// ============================================================================
// COMMENTS
// ============================================================================

// This is a line comment.

/*
This is a
block comment.
*/



// ============================================================================
// LITERALS
// ============================================================================


// ----------------------------------------------------------------------------
// Numbers
// ----------------------------------------------------------------------------

0;              // Zero is a valid number.
5;              // A number evaluates to itself.
5.0;            // Integers and decimals are both numbers.
5.25;           // Numbers can have a fractional part.
-100.9999;      // Numbers can be negative.
1e6;            // Scientific notation for large numbers.
1e-3;           // Scientific notation for small numbers.
0xff;           // Hexadecimal literal (255).
0b1010;         // Binary literal (10).
0o755;          // Octal literal (493).
Infinity;       // Positive infinity.
-Infinity;      // Negative infinity.
NaN;            // Represents "Not a Number".


// ----------------------------------------------------------------------------
// Booleans
// ----------------------------------------------------------------------------

true;           // Boolean true.
false;          // Boolean false.


// ----------------------------------------------------------------------------
// Strings
// ----------------------------------------------------------------------------

"";                     // The empty string.
"Hello";                // Double-quoted string.
'Hello';                // Single-quoted string.
`Hello`;                // Template string using backticks.
"Hello\nWorld";         // Escape sequences insert special characters.


// ----------------------------------------------------------------------------
// Other Primitive Values
// ----------------------------------------------------------------------------

undefined;      // Represents the absence of a value.
null;           // Represents an intentional empty value.



// ============================================================================
// VARIABLES
// ============================================================================

let score = 0;          // Declare a mutable variable.
let lives = 3;          // Variables can be declared independently.

score = 100;            // Variables can be reassigned.
score = score + 1;      // Variables can use their current value.
lives = lives - 1;      // Updating a variable is a common operation.

const width = 800;      // Declare an immutable variable.
const height = 600;     // Constants cannot be reassigned.



// ============================================================================
// ARITHMETIC OPERATORS
// ============================================================================


// ----------------------------------------------------------------------------
// Binary Arithmetic
// ----------------------------------------------------------------------------

5 + 3;          // Addition.
5 - 3;          // Subtraction.
5 * 3;          // Multiplication.
5 / 2;          // Division can produce decimals.
5 % 2;          // Remainder after division.
2 ** 8;         // Exponentiation (2 raised to the power of 8).


// ----------------------------------------------------------------------------
// Unary Arithmetic
// ----------------------------------------------------------------------------

let x = 5;

+x;             // Unary plus converts a value to a number.
-x;             // Unary minus negates a number.

++x;            // Increment before the expression is evaluated.
x++;            // Increment after the expression is evaluated.

--x;            // Decrement before the expression is evaluated.
x--;            // Decrement after the expression is evaluated.


// ----------------------------------------------------------------------------
// Compound Assignment
// ----------------------------------------------------------------------------

let y = 10;

y += 5;         // Equivalent to y = y + 5.
y -= 2;         // Equivalent to y = y - 2.
y *= 3;         // Equivalent to y = y * 3.
y /= 4;         // Equivalent to y = y / 4.
y %= 2;         // Equivalent to y = y % 2.
y **= 3;        // Equivalent to y = y ** 3.


// ============================================================================
// COMPLEX ARITHMETIC EXPRESSIONS
// ============================================================================

2 + 3 * 5;             // Multiplication happens before addition (17).

(2 + 3) * 5;           // Parentheses change the order of evaluation (25).

10 - 2 * 3;            // Multiplication happens before subtraction (4).

(10 - 2) * 3;          // Parentheses force subtraction first (24).

20 / 2 + 5;            // Division happens before addition (15).

20 / (2 + 3);          // Parentheses change the divisor (4).

2 * 3 + 4 * 5;         // Operators of the same precedence are evaluated left to right (26).

(2 * (3 + 4)) * 5;     // Parentheses can be nested.

2 ** 3 * 4;            // Exponentiation happens before multiplication (32).

2 * 3 ** 4;            // Exponentiation has higher precedence than multiplication (162).

-(2 + 3);              // Unary minus applies to the grouped expression (-5).

-2 ** 2;               // Exponentiation happens before unary minus (-4).

(-2) ** 2;             // Parentheses make the negative number the base (4).


// ============================================================================
// COMPARISON OPERATORS
// ============================================================================

5 == "5";       // Loose equality converts values before comparing.
5 != "6";       // Loose inequality also performs type coercion.

5 === 5;        // Strict equality compares both type and value.
5 !== "5";      // Strict inequality compares both type and value.

5 < 10;         // Less than.
5 > 2;          // Greater than.
5 <= 5;         // Less than or equal.
5 >= 5;         // Greater than or equal.



// ============================================================================
// BOOLEAN OPERATORS
// ============================================================================

!true;                  // Logical NOT inverts a boolean.
true && false;          // Logical AND requires both operands to be true.
true || false;          // Logical OR requires at least one operand to be true.

5 > 3 && 10 < 20;       // Boolean expressions can be combined with AND.
5 < 3 || 10 < 20;       // OR succeeds if either expression is true.
!(5 === 5);             // NOT negates the result of an expression.



// ============================================================================
// BITWISE OPERATORS
// ============================================================================

5 & 3;          // Bitwise AND.
5 | 3;          // Bitwise OR.
5 ^ 3;          // Bitwise XOR.
~5;             // Bitwise NOT.

5 << 1;         // Shift bits left by one position.
5 >> 1;         // Arithmetic right shift.
5 >>> 1;        // Logical right shift.



// ============================================================================
// ASSIGNMENT OPERATORS
// ============================================================================

let value = 10;

value = 20;     // Replace the current value.

value &&= 30;   // Assign only if the current value is truthy.
value ||= 40;   // Assign only if the current value is falsy.

let answer = null;

answer ??= 42;  // Assign only if the value is null or undefined.



// ============================================================================
// TYPE OPERATOR
// ============================================================================

typeof 5;               // Evaluates to "number".
typeof true;            // Evaluates to "boolean".
typeof "Hello";         // Evaluates to "string".
typeof undefined;       // Evaluates to "undefined".
typeof null;            // Evaluates to "object" (historical JavaScript quirk).
typeof NaN;             // NaN is still a number.



// ============================================================================
// TRUTHINESS
// ============================================================================

Boolean(true);          // true remains true.
Boolean(false);         // false remains false.

Boolean(1);             // Non-zero numbers are truthy.
Boolean(0);             // Zero is falsy.

Boolean("Hello");       // Non-empty strings are truthy.
Boolean("");            // The empty string is falsy.

Boolean(null);          // null is falsy.
Boolean(undefined);     // undefined is falsy.
Boolean(NaN);           // NaN is falsy.



// ============================================================================
// CONDITIONAL EXPRESSIONS
// ============================================================================

true ? "yes" : "no";    // Chooses between two values.
5 > 3 ? 1 : 0;          // The condition may be any boolean expression.



// ============================================================================
// STRING OPERATIONS
// ============================================================================

"Hello" + " World";                 // Strings can be concatenated.
"Score: " + 10;                     // Non-strings are converted automatically.
`Score: ${10}`;                     // Template strings support interpolation.
"Hello\nWorld";                     // Escape sequences create special characters.
"JavaScript".length;                // Strings have useful properties.
